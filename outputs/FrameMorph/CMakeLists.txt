cmake_minimum_required(VERSION 3.24)
project(FrameMorph VERSION 0.8.0 LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_ARCHITECTURES 86 CACHE STRING "CUDA GPU architectures")
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Threads REQUIRED)
set(FRAMEMORPH_SYSROOT "" CACHE PATH "Optional unpacked Ubuntu dependency root")
find_path(OPENCV_INCLUDE opencv2/core.hpp HINTS "${FRAMEMORPH_SYSROOT}/usr/include/opencv4" PATH_SUFFIXES opencv4 REQUIRED)
find_path(FLTK_INCLUDE FL/Fl.H HINTS "${FRAMEMORPH_SYSROOT}/usr/include" REQUIRED)
find_path(JSON_INCLUDE nlohmann/json.hpp HINTS "${FRAMEMORPH_SYSROOT}/usr/include" REQUIRED)
find_path(FREETYPE_INCLUDE ft2build.h HINTS "${FRAMEMORPH_SYSROOT}/usr/include/freetype2" PATH_SUFFIXES freetype2 REQUIRED)
find_file(LIB_FREETYPE_DYNAMIC NAMES libfreetype.so.6 PATHS "${FRAMEMORPH_SYSROOT}/usr/lib/x86_64-linux-gnu" /usr/lib/x86_64-linux-gnu /lib/x86_64-linux-gnu REQUIRED)
foreach(lib opencv_core opencv_imgproc opencv_imgcodecs fltk)
  find_library(LIB_${lib} NAMES ${lib} HINTS "${FRAMEMORPH_SYSROOT}/usr/lib/x86_64-linux-gnu" REQUIRED)
endforeach()
add_library(framemorph_engine src/engine.cpp src/media.cpp src/drawing.cpp src/editing.cpp src/transforms.cpp src/performance.cpp src/cuda_backend.cu)
target_include_directories(framemorph_engine PUBLIC src ${OPENCV_INCLUDE} ${JSON_INCLUDE} PRIVATE ${FREETYPE_INCLUDE})
target_link_libraries(framemorph_engine PUBLIC ${LIB_opencv_core} ${LIB_opencv_imgproc} ${LIB_opencv_imgcodecs} Threads::Threads PRIVATE ${LIB_FREETYPE_DYNAMIC})
target_compile_definitions(framemorph_engine PRIVATE FRAMEMORPH_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(framemorph_engine PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic> $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr>)
add_executable(framemorph src/main.cpp src/asset_editor.cpp src/editor.cpp src/sample.cpp)
target_include_directories(framemorph PRIVATE ${FLTK_INCLUDE})
target_compile_definitions(framemorph PRIVATE FRAMEMORPH_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(framemorph PRIVATE framemorph_engine ${LIB_fltk})
add_executable(framemorph_tests tests/tests.cpp src/sample.cpp)
target_link_libraries(framemorph_tests PRIVATE framemorph_engine)
set_target_properties(framemorph framemorph_tests PROPERTIES
  BUILD_WITH_INSTALL_RPATH TRUE
  INSTALL_RPATH "$ORIGIN/../lib")
enable_testing()
add_test(NAME engine COMMAND framemorph_tests)
set_tests_properties(engine PROPERTIES TIMEOUT 180)
install(TARGETS framemorph RUNTIME DESTINATION bin)
